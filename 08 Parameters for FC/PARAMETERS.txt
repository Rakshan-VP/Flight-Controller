-----
0 = NONE        // no mode / uninitialized
1 = STABILIZE   // self-level (angle control)
2 = ALT_HOLD    // maintain altitude (barometer or GPS)
3 = POS_HOLD    // hold position (GPS)
4 = LOITER      // smooth position hold

5 = RTL         // return to launch
6 = LAND        // controlled descent
7 = TAKEOFF     // auto takeoff

8 = CUT         // immediate motor stop (emergency)
